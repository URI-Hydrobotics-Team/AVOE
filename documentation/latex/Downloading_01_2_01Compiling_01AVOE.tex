\begin{DoxyCode}{0}
\DoxyCodeLine{-\/\ gcc/g++}
\DoxyCodeLine{-\/\ make\ and\ cmake\ (for\ cmake\ build\ option)}

\end{DoxyCode}

Make sure you are in the core directory in the AVOE repository 
\begin{DoxyCode}{0}
\DoxyCodeLine{\string~/AVOE/core/\ }

\end{DoxyCode}
 to compile enter the following command see below for the options 
\begin{DoxyCode}{0}
\DoxyCodeLine{/build.sh\ <option>\ }

\end{DoxyCode}
 \hypertarget{Downloading_01_2_01Compiling_01AVOE_autotoc_md17}{}\doxysubsubsection{\texorpdfstring{Compilation options}{Compilation options}}\label{Downloading_01_2_01Compiling_01AVOE_autotoc_md17}
There are options when you compile


\begin{DoxyTable}{2}{}{}{1}
\SetCell{c,bg=\tableheadbgcolor,font=\bfseries}Option  &\SetCell{c,bg=\tableheadbgcolor,font=\bfseries}Description  \\
virtual  &compiles program using dummy data  \\
virtual\+\_\+verbose  &Similar to virtual with verbose debug messages  \\
virtual\+\_\+g  &Builds virtual target with the Address Sanitizer  \\
physical  &Builds the physical AVOE target with real world data  \\
\end{DoxyTable}
